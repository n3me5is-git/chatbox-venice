cmake_minimum_required(VERSION 3.22)
project(chatbox_paddle_ocr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ncnn REQUIRED)
find_package(OpenCV REQUIRED)
include(FetchContent)
FetchContent_Declare(clipper URL https://github.com/AngusJohnson/Clipper2/archive/21ebba05db8894f0c7217ad35ea518080f324946.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(clipper)
if(NOT clipper_POPULATED)
  FetchContent_Populate(clipper)
endif()
add_library(ocr_core STATIC core.cpp
  ${clipper_SOURCE_DIR}/CPP/Clipper2Lib/src/clipper.engine.cpp
  ${clipper_SOURCE_DIR}/CPP/Clipper2Lib/src/clipper.offset.cpp
  ${clipper_SOURCE_DIR}/CPP/Clipper2Lib/src/clipper.rectclip.cpp)
target_include_directories(ocr_core PUBLIC ${OpenCV_INCLUDE_DIRS} ${JSON_INCLUDE_DIR} ${clipper_SOURCE_DIR}/CPP/Clipper2Lib/include)
target_link_libraries(ocr_core PUBLIC ncnn ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(ocr_core PRIVATE /utf-8)
endif()
if(ANDROID)
  add_library(paddle_ocr SHARED jni.cpp)
  target_link_libraries(paddle_ocr PRIVATE ocr_core android jnigraphics log)
else()
  add_executable(paddle-ocr-worker main.cpp)
  target_link_libraries(paddle-ocr-worker PRIVATE ocr_core)
  if(MSVC)
    target_compile_options(paddle-ocr-worker PRIVATE /utf-8)
  endif()
  install(TARGETS paddle-ocr-worker RUNTIME DESTINATION .)
endif()
